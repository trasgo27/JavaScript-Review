Módulo 00: XMLHttpRequest

Ejercicio 02 — GET con manejo de eventos

XMLHttpRequest tiene varios eventos: onload, onerror, onreadystatechange. Practica usarlos.

Tasks

Task 1: Usar onreadystatechange

En vez de onload, usa onreadystatechange y verifica que readyState === 4 y status === 200.

Task 2: Mostrar cada post

Fetch a /posts?_limit=5 y muestra cada título con un for loop.

Task 3: Mostrar en el DOM

Crea un <ul> con los títulos de los posts. Usa innerHTML.

💡 Ver solución completa (click para expandir) 
// ══════════════════════════════════════════════════════════════
// SOLUCIÓN Ejercicio 02 — GET con eventos
// ══════════════════════════════════════════════════════════════

console.log('=== Ejercicio 02: GET con eventos ===');

// ─── TASK 1: onreadystatechange ─────────────────────────────
hacerEj('TASK 1: Usar onreadystatechange', () => {
    const xhr = new XMLHttpRequest();
    xhr.open('GET', 'https://jsonplaceholder.typicode.com/posts/1');

    xhr.onreadystatechange = function() {
        // readyState values: 0=UNSENT, 1=OPENED, 2=HEADERS_RECEIVED, 3=LOADING, 4=DONE
        if (xhr.readyState === 4 && xhr.status === 200) {
            const datos = JSON.parse(xhr.responseText);
            console.log('readyState=4, status=200. Título:', datos.title);
        }
    };

    xhr.send();
});

// ─── TASK 2: Mostrar cada post ──────────────────────────────
hacerEj('TASK 2: Listar 5 posts', () => {
    const xhr = new XMLHttpRequest();
    xhr.open('GET', 'https://jsonplaceholder.typicode.com/posts?_limit=5');

    xhr.onload = function() {
        const posts = JSON.parse(xhr.responseText);
        console.log(`Se recibieron ${posts.length} posts:`);

        for (let i = 0; i < posts.length; i++) {
            console.log(`  ${i + 1}. ${posts[i].title}`);
        }
    };

    xhr.send();
});

// ─── TASK 3: Mostrar en DOM ─────────────────────────────────
hacerEj('TASK 3: Renderizar en el DOM', () => {
    const xhr = new XMLHttpRequest();
    xhr.open('GET', 'https://jsonplaceholder.typicode.com/posts?_limit=5');

    xhr.onload = function() {
        const posts = JSON.parse(xhr.responseText);

        // Crear un ul con los títulos
        let html = '<ul>';
        for (const post of posts) {
            html += `<li>${post.title}</li>`;
        }
        html += '</ul>';

        // Insertar en el DOM
        const container = document.getElementById('output');
        if (container) container.innerHTML = html;
    };

    xhr.send();
});

function hacerEj(taskName, taskFunction) {
    console.log(`\n--- ${taskName} ---`);
    try { taskFunction(); } catch (error) { console.error(`Error:`, error); }
}
