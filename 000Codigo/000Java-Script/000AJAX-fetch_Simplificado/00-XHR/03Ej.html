Módulo 00: XMLHttpRequest

Ejercicio 03 — POST con XHR

Enviar datos al servidor también es posible con XHR. Necesitas configurar method, headers y body.

xhr.open('POST', 'URL');
xhr.setRequestHeader('Content-type', 'application/json; charset=UTF-8');
xhr.onload = function() { console.log(JSON.parse(xhr.responseText)); };
xhr.send(JSON.stringify({ title: 'foo', body: 'bar', userId: 1 }));

Tasks

Task 1: Configurar un POST

Abre una conexión POST a /posts y configura el header Content-type.

Task 2: Enviar datos

Crea un objeto, conviértelo con JSON.stringify() y envíalo con xhr.send().

Task 3: Comparar con fetch POST

Escribe la misma petición con fetch() para ver cuánto más limpio es.

// ─── TASK 1 & 2: POST completo ──────────────────────────────
hacerEj('TASK 1 & 2: POST con XHR', () => {

    const datos = {
        title: 'Mi post con XHR',
        body: 'Estoy aprendiendo XMLHttpRequest',
        userId: 1
    };

    const xhr = new XMLHttpRequest();
    xhr.open('POST', 'https://jsonplaceholder.typicode.com/posts');

    // 💡 Pista: setRequestHeader necesita dos argumentos:
    //   - Nombre del header: 'Content-type'
    //   - Valor: 'application/json; charset=UTF-8'
    xhr.setRequestHeader('Content-type', 'application/json; charset=UTF-8');

    xhr.onload = function() {
        const respuesta = JSON.parse(xhr.responseText);
        console.log('Post creado con XHR:', respuesta);
        console.log('ID asignado:', respuesta.id);
    };

    xhr.onerror = function() {
        console.error('Error de red');
    };

    // 💡 Pista: send() necesita el body como string:
    //   xhr.send(JSON.stringify(datos));
    /* Tu código aquí */

});

// ─── TASK 3: Comparar con fetch ─────────────────────────────
hacerEj('TASK 3: Comparar con fetch POST', async () => {

    console.log('\n--- Con XMLHttpRequest (mucho código): ---');
    console.log('  xhr.open("POST", URL)');
    console.log('  xhr.setRequestHeader("Content-type", "...")');
    console.log('  xhr.onload = function() { ... }');
    console.log('  xhr.send(JSON.stringify(data))');

    console.log('\n--- Con fetch (mucho menos código): ---');

    // La misma petición con fetch:
    const response = await fetch('https://jsonplaceholder.typicode.com/posts', {
        method: 'POST',
        headers: { 'Content-type': 'application/json; charset=UTF-8' },
        body: JSON.stringify({
            title: 'Mi post con fetch',
            body: 'fetch es más limpio',
            userId: 1
        })
    });
    const result = await response.json();
    console.log('  fetch(url, { method, headers, body })');
    console.log('Post creado con fetch:', result);
    console.log('ID asignado:', result.id);
    console.log('\n✅ fetch() es más legible y moderno que XHR');

});

function hacerEj(taskName, taskFunction) {
    console.log(`\n--- ${taskName} ---`);
    try { taskFunction(); } catch (error) { console.error(`Error:`, error); }
}
