Ejercicio 1 Fácil

Store simple con signals en un servicio.

@Injectable({ providedIn: 'root' })
export class CartStore {
  items = signal<Product[]>([]);
  total = computed(() => this.items().reduce((s, i) => s + i.price * i.qty, 0));

  add(p: Product) {
    this.items.update(list => {
      const existing = list.find(i => i.id === p.id);
      if (existing) return list.map(i => i.id === p.id ? { ...i, qty: i.qty + 1 } : i);
      return [...list, { ...p, qty: 1 }];
    });
  }

  remove(id: number) {
    this.items.update(list => list.filter(i => i.id !== id));
  }
}

Ejercicio 2 Medio

NgRx SignalStore básico.

import { signalStore, withState, withMethods, patchState } from '@ngrx/signals';

export const CounterStore = signalStore(
  { providedIn: 'root' },
  withState({ count: 0 }),
  withMethods((store) => ({
    increment() { patchState(store, { count: store.count() + 1 }); },
    decrement() { patchState(store, { count: store.count() - 1 }); },
    reset() { patchState(store, { count: 0 }); }
  }))
);

// Componente:
store = inject(CounterStore);

Ejercicio 3 Medio

SignalStore con efectos (rxMethods).

import { rxMethod } from '@ngrx/signals/rxjs-interop';
import { tapResponse } from '@ngrx/operators';

export const UserStore = signalStore(
  withState<{ users: User[]; loading: boolean }>({ users: [], loading: false }),
  withMethods((store, http = inject(HttpClient)) => ({
    loadUsers: rxMethod<void>(
      pipe(
        tap(() => patchState(store, { loading: true })),
        switchMap(() => http.get<User[]>('/api/users')),
        tapResponse({
          next: (users) => patchState(store, { users, loading: false }),
          error: () => patchState(store, { loading: false })
        })
      )
    )
  }))
);

Ejercicio 4 Difícil

Persistencia en localStorage con effect.

@Injectable({ providedIn: null })
export class ThemeStore {
  theme = signal<'light' | 'dark'>(
    (localStorage.getItem('theme') as any) || 'light'
  );

  constructor() {
    effect(() => {
      localStorage.setItem('theme', this.theme());
      document.body.classList.toggle('dark', this.theme() === 'dark');
    });
  }

  toggle() {
    this.theme.update(t => t === 'light' ? 'dark' : 'light');
  }
}

Ejercicio 5 Difícil

Feature store con SignalStore (módulos separados).

export const AuthStore = signalStore(
  { providedIn: 'root' },
  withState({ user: null as User | null, token: '' }),
  withComputed((state) => ({
    isLoggedIn: computed(() => !!state.user()),
    displayName: computed(() => state.user()?.name || 'Invitado')
  })),
  withMethods((store, auth = inject(AuthService)) => ({
    login: rxMethod<Credentials>(
      pipe(
        switchMap(c => auth.login(c)),
        tapResponse({
          next: (res) => patchState(store, { user: res.user, token: res.token }),
          error: console.error
        })
      )
    ),
    logout() { patchState(store, { user: null, token: '' }); }
  }))
);
