13 · Routing & Guards Avanzado

El router de Angular gestiona la navegación entre vistas. Incluye lazy loading, parámetros, guards y resolvers.

Ejercicios

Ejercicio 1 Fácil

Configuración básica de rutas.

import { Routes } from '@angular/router';

export const routes: Routes = [
  { path: '', component: HomeComponent },
  { path: 'about', component: AboutComponent },
  { path: '**', component: NotFoundComponent }
];

// app.config.ts:
export const appConfig: ApplicationConfig = {
  providers: [provideRouter(routes)]
};

Ejercicio 3 Medio

Rutas con parámetros.

// Ruta:
{ path: 'user/:id', component: UserComponent }

// Componente:
route = inject(ActivatedRoute);
userId = '';

ngOnInit() {
  this.route.params.subscribe(p => this.userId = p['id']);
}

// Navegación:
// <a [routerLink]="['/user', 42]">User 42</a>

Ejercicio 4 Medio

Lazy loading de módulos/componentes.

export const routes: Routes = [
  {
    path: 'admin',
    loadComponent: () => import('./admin/admin.component')
      .then(m => m.AdminComponent)
  },
  {
    path: 'dashboard',
    loadChildren: () => import('./dashboard/routes')
      .then(m => m.DASHBOARD_ROUTES)
  }
];

Ejercicio 5 Difícil

Guard canActivate funcional.

export const authGuard: CanActivateFn = (route, state) => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.isLoggedIn()) return true;
  return router.parseUrl('/login');
};

// Uso:
{ path: 'perfil', component: PerfilComponent, canActivate: [authGuard] }

Ejercicio 6 Difícil

Resolver para precargar datos.

export const userResolver: ResolveFn<User> = (route) => {
  const userService = inject(UserService);
  return userService.getUser(route.params['id']);
};

{
  path: 'user/:id',
  component: UserComponent,
  resolve: { user: userResolver }
}

// Componente:
ngOnInit() {
  this.user = this.route.snapshot.data['user'];
}
