14 · HttpClient & Interceptors Avanzado

Realiza peticiones HTTP con HttpClient y personaliza el comportamiento con interceptores.

Ejercicios

Ejercicio 1 Fácil

GET básico.

// app.config.ts:
providers: [provideHttpClient()]

// Servicio:
http = inject(HttpClient);

getUsers() {
  return this.http.get<User[]>('https://api.example.com/users');
}

// Componente:
users = signal<User[]>([]);
ngOnInit() {
  this.userService.getUsers().subscribe(u => this.users.set(u));
}

Ejercicio 2 Fácil

POST con body.

crearUser(user: User) {
  return this.http.post<User>('/api/users', user);
}

// Uso:
this.userService.crearUser(nuevoUser).subscribe(u => {
  console.log('Creado:', u);

Ejercicio 3 Medio

Manejo de errores con catchError.

import { catchError, throwError } from 'rxjs';

getUsers() {
  return this.http.get<User[]>('/api/users').pipe(
    catchError(err => {
      console.error('Error:', err);
      return throwError(() => new Error('No se pudieron cargar los usuarios'));
    })
  );
}

Ejercicio 4 Medio

Interceptor funcional (añade token).

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = localStorage.getItem('token');
  if (token) {
    req = req.clone({
      setHeaders: { Authorization: `Bearer ${token}` }
    });
  }
  return next(req);
};

// app.config.ts:
providers: [provideHttpClient(withInterceptors([authInterceptor]))]

Ejercicio 5 Difícil

Interceptor de logging + manejo global de errores.

export const loggingInterceptor: HttpInterceptorFn = (req, next) => {
  console.log(`→ ${req.method} ${req.url}`);
  const start = Date.now();

  return next(req).pipe(
    tap({
      next: (event) => {
        if (event instanceof HttpResponse) {
          console.log(`← ${req.url} (${Date.now() - start}ms)`);
        }
      },
      error: (err) => console.error(`✗ ${req.url}:`, err)
    })
  );
};
