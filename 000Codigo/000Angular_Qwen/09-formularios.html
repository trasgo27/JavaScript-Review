09 · Formularios Intermedio

Dos enfoques: Template-driven (simple, con ngModel) y Reactive Forms (escalable, con FormGroup/FormBuilder).

Ejercicios

Ejercicio 1 Fácil

Formulario Template-driven básico.

// Importa FormsModule
nombre = '';

enviar() {
  console.log(`Hola ${this.nombre}`);
}

// Template:
// <form (ngSubmit)="enviar()">
//   <input name="nombre" [(ngModel)]="nombre" required />
//   <button type="submit">Enviar</button>

Ejercicio 2 Fácil

Reactive Form con FormBuilder.

import { FormBuilder, Validators, ReactiveFormsModule } from '@angular/forms';

fb = inject(FormBuilder);
form = this.fb.group({
  email: ['', [Validators.required, Validators.email]],
  password: ['', [Validators.required, Validators.minLength(6)]]
});

enviar() {
  if (this.form.valid) console.log(this.form.value);
}

Ejercicio 4 Medio

FormArray dinámico (lista de teléfonos).

form = this.fb.group({
  nombre: ['', Validators.required],
  telefonos: this.fb.array([this.fb.control('')])
});

get telefonos() { return this.form.get('telefonos') as FormArray; }

agregarTelefono() { this.telefonos.push(this.fb.control('')); }

// Template:
// @for (tel of telefonos.controls; track $index) {
//   <input [formControl]="$any(tel)" />

Ejercicio 5 Difícil

Validador personalizado (confirmar contraseña).

function passwordMatchValidator(g: FormGroup) {
  const p = g.get('password')?.value;
  const c = g.get('confirm')?.value;
  return p === c ? null : { mismatch: true };
}

form = this.fb.group({
  password: ['', Validators.required],
  confirm: ['', Validators.required]
}, { validators: passwordMatchValidator });
