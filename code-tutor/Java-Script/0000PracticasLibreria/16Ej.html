Setup (simulated APIs)

// 75% success rate API
function fetchData(id) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            Math.random() > 0.25
                ? resolve({ id, data: `Result for ${id}` })
                : reject(new Error(`Failed: ${id}`));
        }, 200 + Math.random() * 300);
    });
}

// 50% success rate (harder)
function fragileFetch(id) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            Math.random() > 0.5
                ? resolve({ id, data: `Fragile result ${id}` })
                : reject(new Error(`Fragile failed: ${id}`));
        }, 100 + Math.random() * 200);
    });
}

// Fetch all with allSettled
function fetchAll(ids, fetcher) {
    return Promise.allSettled(ids.map(id => fetcher(id)));
}

Reference (the retry pattern)

function retry(items, fetcher, maxRetries, onRetry) {
    const MAX = maxRetries ?? 3;

    function attempt(list, results, tries) {
        if (list.length === 0 || tries > MAX) {
            console.log(`Done after ${tries - 1} attempt(s)`);
            return results;
        }

        if (onRetry) onRetry(list, tries);

        return fetchAll(list, fetcher).then(resolved => {
            const newSuccesses = resolved
                .filter(r => r.status === 'fulfilled')
                .map(r => r.value);

            const newFailures = resolved.reduce((acc, r, i) => {
                if (r.status === 'rejected') acc.push(list[i]);
                return acc;
            }, []);

            return attempt(newFailures, results.concat(newSuccesses), tries + 1);
        });
    }

    return attempt(items, [], 1);
}
