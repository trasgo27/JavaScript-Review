03 — CREATE TABLE & INSERT

Design tables with constraints, insert data, and handle errors.

Database file: school.db — start fresh each time with a new database.

Step 0: Open

sqlite3 school.db
.mode column
.headers on

Task 1: Create students table

CREATE TABLE students (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  enrolled INTEGER DEFAULT 1
);

Task 2: Verify the Schema

.schema students
.tables

Task 3: INSERT Single Row

INSERT INTO students (name, email) VALUES ('Alice', 'alice@school.com');
SELECT * FROM students;

Task 4: INSERT Multiple Rows

INSERT INTO students (name, email, enrolled) VALUES
  ('Bob', 'bob@school.com', 1),
  ('Charlie', 'charlie@school.com', 1),
  ('Diana', 'diana@school.com', 0);
SELECT * FROM students;

Task 5: Test NOT NULL Constraint

-- This will fail — name is required
INSERT INTO students (email) VALUES ('no-name@school.com');
Runtime error: NOT NULL constraint failed: students.name

Task 6: Test UNIQUE Constraint

-- This will fail — alice@school.com already exists
INSERT INTO students (name, email) VALUES ('Alice Clone', 'alice@school.com');
Runtime error: UNIQUE constraint failed: students.email

Task 7: Test DEFAULT Value

INSERT INTO students (name, email) VALUES ('Eve', 'eve@school.com');
SELECT name, enrolled FROM students WHERE enrolled = 1;
-- Eve is enrolled=1 even though we didn't specify it

Task 8: Create a courses table (foreign key)

CREATE TABLE courses (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  student_id INTEGER,
  FOREIGN KEY (student_id) REFERENCES students(id)
);

INSERT INTO courses (title, student_id) VALUES
  ('Math', 1),
  ('Science', 2),
  ('History', 1);
SELECT * FROM courses;
Note: By default SQLite does not enforce foreign keys. Enable with PRAGMA foreign_keys = ON; before creating tables.
