05 — ORDER BY, LIMIT & Aggregates

Sort results, limit rows, and calculate totals with aggregate functions.

Database file: store.db — same seed from Exercise 02.

Step 0: Open store.db

sqlite3 store.db
.mode column
.headers on

Task 1: ORDER BY (Ascending)

SELECT name, price FROM products ORDER BY price;

Task 2: ORDER BY (Descending)

SELECT name, price FROM products ORDER BY price DESC;

Task 3: LIMIT (Top N)

-- Top 3 most expensive products
SELECT name, price FROM products ORDER BY price DESC LIMIT 3;

Task 4: LIMIT with OFFSET (Pagination)

-- Skip first 2, show next 3
SELECT name, price FROM products ORDER BY price DESC LIMIT 3 OFFSET 2;

Task 5: COUNT (Total Rows)

SELECT COUNT(*) AS total_users FROM users;
SELECT COUNT(*) AS total_products FROM products;

Task 6: SUM, AVG, MIN, MAX

SELECT
  COUNT(*) AS total_items,
  SUM(stock) AS total_stock,
  AVG(price) AS avg_price,
  MIN(price) AS cheapest,
  MAX(price) AS most_expensive
FROM products;

Task 7: GROUP BY (Per Category)

SELECT
  category,
  COUNT(*) AS items,
  AVG(price) AS avg_price,
  SUM(stock) AS total_stock
FROM products
GROUP BY category;

Task 8: HAVING (Filter Groups)

-- Categories where average price > 50
SELECT
  category,
  AVG(price) AS avg_price
FROM products
GROUP BY category
HAVING avg_price > 50;

Task 9: Youngest User

SELECT name, age FROM users ORDER BY age ASC LIMIT 1;

Task 10: ROUND for Clean Numbers

SELECT category, ROUND(AVG(price), 2) AS avg_price
FROM products
GROUP BY category;
